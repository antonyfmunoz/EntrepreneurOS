**Part A: Create `shared/pipeline-schemas.ts`**

Define Zod schemas for pipeline state contracts per D-08 (modular, composable, each phase defines its own I/O shape):

```typescript
import { z } from "zod";

export const PageStateSchema = z.object({
  pageName: z.string(),
  pageIndex: z.number().int().min(0),
  status: z.enum(["pending", "in_progress", "complete", "failed"]),
  error: z.string().nullable().default(null),
  output: z.unknown().nullable().default(null),
});

export const ProjectConfigSchema = z.object({
  projectId: z.string().min(1),
  repoPath: z.string().min(1),
  framework: z.enum(["react-vite-tailwind-shadcn"]),
  stitchProjectId: z.string().optional(),
});

export const PipelineRunSchema = z.object({
  projectId: z.string().min(1),
  phase: z.enum(["spec", "ui-gen", "integration", "backend", "deploy"]),
  status: z.enum(["running", "paused", "complete", "failed"]).default("running"),
  config: ProjectConfigSchema,
});

// Phase-specific output schemas (D-08: each phase defines its own)
export const SpecPhaseOutputSchema = z.object({
  pages: z.array(z.object({
    name: z.string(),
    purpose: z.string(),
    components: z.array(z.string()),
    dataRequirements: z.array(z.string()),
  })),
});

export const UiGenPhaseOutputSchema = z.object({
  htmlUrl: z.string().url(),
  screenshotUrl: z.string().url(),
  tokenVersion: z.number().int(),
  approved: z.boolean(),
});

// Type exports
export type PageState = z.infer<typeof PageStateSchema>;
export type ProjectConfig = z.infer<typeof ProjectConfigSchema>;
export type PipelineRun = z.infer<typeof PipelineRunSchema>;
export type SpecPhaseOutput = z.infer<typeof SpecPhaseOutputSchema>;
export type UiGenPhaseOutput = z.infer<typeof UiGenPhaseOutputSchema>;
```
