    ```typescript
    import { GoogleGenerativeAI } from "@google/generative-ai";
    import { z } from "zod";
    import { ReviewScoreSchema } from "./types.js";
    import type { ReviewScore, DmTokenRow } from "./types.js";
    import type { PageSpecFull } from "@shared/spec-schema.js";

    export interface GeminiReviewInput {
      screenshotUrls: string[];
      spec: PageSpecFull;
      tokens: DmTokenRow | null;
      priorPatterns: Array<{ name: string; usageContext?: string | null }>;
    }

    /**
     * Secondary vision-based reviewer using Gemini 2.0 Pro.
     * Evaluates Stitch screenshots (not HTML) against the same 4 dimensions.
     * Returns null if GEMINI_API_KEY not set or on error — non-blocking.
     */
    export async function geminiReview(
      input: GeminiReviewInput
    ): Promise<ReviewScore | null> {
      const apiKey = process.env.GEMINI_API_KEY;
      if (!apiKey) {
        return null;
      }

      try {
        const genAI = new GoogleGenerativeAI(apiKey);
        const model = genAI.getGenerativeModel({ model: "gemini-2.0-pro" });

        // Build review prompt with the same 4 dimensions as Claude
        const prompt = buildGeminiReviewPrompt(input);

        // Gemini 2.0 Pro can accept image URLs in multipart content
        // For now, describe the screenshots in text and ask for vision-informed review
        // The screenshot URLs are Stitch presigned URLs — Gemini can fetch them
        const parts: Array<{ text: string }> = [{ text: prompt }];

        const result = await model.generateContent({
          contents: [{ role: "user", parts }],
          generationConfig: {
            responseMimeType: "application/json",
          },
        });

        const responseText = result.response.text();
        const parsed = JSON.parse(responseText);
        return ReviewScoreSchema.parse(parsed);
      } catch {
        // Gemini error — return null (non-blocking)
        return null;
      }
    }

    function buildGeminiReviewPrompt(input: GeminiReviewInput): string {
      const { spec, tokens, priorPatterns, screenshotUrls } = input;

      const sections: string[] = [];

      sections.push("You are a UI quality reviewer. Evaluate the generated UI screenshots against the specification.");
      sections.push(`\nPage: ${spec.name}\nPurpose: ${spec.purpose}\nComponents required: ${spec.components.join(", ")}`);

      if (tokens) {
        const tokenSummary = [
          tokens.colorPrimary ? `primary: ${tokens.colorPrimary}` : null,
          tokens.typeFontFamily ? `font: ${tokens.typeFontFamily}` : null,
          tokens.borderRadius !== null ? `radius: ${tokens.borderRadius}px` : null,
        ].filter(Boolean).join(", ");
        if (tokenSummary) sections.push(`\nDesign tokens: ${tokenSummary}`);
      }

      if (priorPatterns.length > 0) {
        sections.push(`\nEstablished patterns: ${priorPatterns.map(p => p.name).join(", ")}`);
      }

      sections.push(`\nScreenshot URLs for review:\n${screenshotUrls.map((u, i) => `  ${i + 1}. ${u}`).join("\n")}`);

      sections.push(`\nScore each dimension 0.0-1.0 with findings. Return JSON:
    {
      "specCompliance": { "score": 0.0-1.0, "findings": ["..."] },
      "visualConsistency": { "score": 0.0-1.0, "findings": ["..."] },
      "structuralCompleteness": { "score": 0.0-1.0, "findings": ["..."] },
      "contentQuality": { "score": 0.0-1.0, "findings": ["..."] }
    }`);

      return sections.join("\n");
    }
    ```

    **Step 3: Create lib/ui-generator/targeted-refiner.ts**

    ```typescript
    import type { RefineResult } from "./types.js";

    /**
     * Uses 21st.dev component refiner MCP tool to fix a specific component
     * instead of re-calling Stitch for the full page.
     *
     * @param componentName - Name of the component to refine
     * @param existingCode - Current component code from Stitch output
     * @param instructions - What to fix (from review findings or user feedback)
     * @param mcpInvoke - Injectable MCP invocation function (same pattern as component-discovery)
     * @returns RefineResult with refined code, or null if refinement not available
     */
    export async function refineComponent(input: {
      componentName: string;
      existingCode: string;
      instructions: string;
      mcpInvoke?: (toolName: string, args: Record<string, unknown>) => Promise<unknown>;
    }): Promise<RefineResult | null> {
      const { componentName, existingCode, instructions, mcpInvoke } = input;

      if (!mcpInvoke) {
        return null;
      }

      try {
        const result = await mcpInvoke(
          "mcp__magic21__21st_magic_component_refiner",
          {
            code: existingCode,
            instructions: `Refine the ${componentName} component: ${instructions}`,
          }
        );

        if (result && typeof result === "object" && typeof (result as any).code === "string") {
          return {
            componentName,
            refinedCode: (result as any).code as string,
            source: "21st-dev-refiner",
          };
        }

        return null;
      } catch {
        // MCP refiner not available or error — return null
        return null;
      }
    }

    /**
     * Determines if a rejection is component-level (fixable by targeted refinement)
     * or page-level (requires full Stitch re-call).
     *
     * Component-level: one or two specific findings mention a component by name
     * Page-level: layout issues, overall structure, color scheme, general feedback
     */
    export function isComponentLevelIssue(
      failedDimensions: string[],
      findings: string[],
      componentNames: string[]
    ): { isComponentLevel: boolean; targetComponents: string[] } {
      // If layout/structure dimensions failed, it's page-level
      if (failedDimensions.includes("structuralCompleteness")) {
        return { isComponentLevel: false, targetComponents: [] };
      }

      // Check if findings mention specific component names
      const mentionedComponents: string[] = [];
      for (const finding of findings) {
        const lower = finding.toLowerCase();
        for (const name of componentNames) {
          if (lower.includes(name.toLowerCase()) && !mentionedComponents.includes(name)) {
            mentionedComponents.push(name);
          }
        }
      }

      // Component-level if 1-2 specific components are mentioned
      if (mentionedComponents.length > 0 && mentionedComponents.length <= 2) {
        return { isComponentLevel: true, targetComponents: mentionedComponents };
      }

      return { isComponentLevel: false, targetComponents: [] };
    }
    ```

    ```typescript
    /**
     * Combines Claude text-based review with Gemini vision-based review.
     * Returns DualReviewScore with combined worst-of-both per dimension.
     *
     * The existing selfReview function is preserved for backwards compatibility.
     * dualReview calls selfReview internally, then calls geminiReview in parallel.
     */
    export async function dualReview(input: SelfReviewInput): Promise<DualReviewScore> {
      // Run Claude and Gemini reviews in parallel
      const [claudeScore, geminiScore] = await Promise.all([
        selfReview(input),
        geminiReview({
          screenshotUrls: input.screenshotUrls,
          spec: input.spec,
          tokens: input.tokens,
          priorPatterns: input.priorPatterns,
        }),
      ]);

      // Combine scores — take the LOWER score per dimension from either reviewer
      // This ensures if either reviewer flags an issue, it triggers escalation
      const combined = combineScores(claudeScore, geminiScore);

      return {
        claude: claudeScore,
        gemini: geminiScore,
        combined,
        reviewerCount: geminiScore !== null ? 2 : 1,
      };
    }

    /**
     * Takes the minimum score per dimension from both reviewers.
     * Merges findings from both. If gemini is null, returns claude scores unchanged.
     */
    function combineScores(
      claude: ReviewScore,
      gemini: ReviewScore | null
    ): ReviewScore {
      if (gemini === null) {
        return claude;
      }

      const dimensions = [
        "specCompliance",
        "visualConsistency",
        "structuralCompleteness",
        "contentQuality",
      ] as const;

      const combined: Record<string, { score: number; findings: string[] }> = {};

      for (const dim of dimensions) {
        combined[dim] = {
          score: Math.min(claude[dim].score, gemini[dim].score),
          findings: [
            ...claude[dim].findings.map(f => `[Claude] ${f}`),
            ...gemini[dim].findings.map(f => `[Gemini] ${f}`),
          ],
        };
      }

      return ReviewScoreSchema.parse(combined);
    }
    ```
