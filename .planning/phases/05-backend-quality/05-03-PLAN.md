    **tests/integration/helpers/test-db.ts** — Per D-13:

    Transaction rollback isolation helper:
    ```typescript
    import postgres from "postgres";
    import { drizzle } from "drizzle-orm/postgres-js";

    const testSql = postgres(process.env.DATABASE_URL as string);

    class RollbackSentinel extends Error {
      constructor() { super("ROLLBACK_SENTINEL"); this.name = "RollbackSentinel"; }
    }

    export async function withTestTransaction<T>(
      fn: (db: ReturnType<typeof drizzle>) => Promise<T>
    ): Promise<T | undefined> {
      try {
        return await testSql.begin(async (txSql) => {
          const txDb = drizzle(txSql);
          const result = await fn(txDb);
          throw new RollbackSentinel();
        });
      } catch (err) {
        if (err instanceof RollbackSentinel) return undefined;
        throw err;
      }
    }

    export async function closeTestConnection(): Promise<void> {
      await testSql.end();
    }
    ```
    Uses a unique RollbackSentinel error class (not string message) per Pitfall 3 from RESEARCH.md to prevent accidental catch-and-swallow.

    Create lib/test-runner/fix-loop.ts — Per D-10, D-11, D-14:

    **`runWithFixLoop(options: { projectRoot: string; testGlob: string; fixFn: (output: string, cycle: number) => Promise<boolean>; maxCycles?: number }): Promise<FixLoopResult>`**

    Implementation:
    ```
    const MAX_CYCLES = options.maxCycles ?? 3;
    const attemptsLog: string[] = [];

    for (let cycle = 1; cycle <= MAX_CYCLES; cycle++) {
      let output = "";
      try {
        // Run vitest with the specific test glob
        output = execSync(
          `npx vitest run ${options.testGlob} --reporter=verbose`,
          { cwd: options.projectRoot, stdio: "pipe", encoding: "utf-8" }
        );
        return { passed: true, cycles: cycle, lastOutput: output };
      } catch (err: any) {
        output = (err.stdout ?? "") + (err.stderr ?? "");
        const summary = parseFailingSummary(output);
        attemptsLog.push(`Cycle ${cycle}: ${summary}`);

        if (cycle < MAX_CYCLES) {
          // D-11: fixFn fixes IMPLEMENTATION, never tests
          const fixed = await options.fixFn(output, cycle);
          if (!fixed) break; // early escalation
        }
      }
    }

    return {
      passed: false,
      cycles: attemptsLog.length,
      lastOutput: attemptsLog[attemptsLog.length - 1] ?? "",
      escalationReport: {
        failingTest: parseFailingTestName(attemptsLog[attemptsLog.length - 1] ?? ""),
        errorMessage: parseErrorMessage(attemptsLog[attemptsLog.length - 1] ?? ""),
        attemptsLog,
        hypothesis: generateHypothesis(attemptsLog),
      },
    };
    ```

    **Helper functions (not exported):**

    `parseFailingSummary(output: string): string`
    - Extract first FAIL line from vitest verbose output (pattern: `FAIL  tests/...`)
    - If no FAIL line found, return first 200 chars of output

    `parseFailingTestName(summary: string): string`
    - Extract test file path from FAIL line

    `parseErrorMessage(summary: string): string`
    - Extract error message after "Error:" or "expected" keywords

    `generateHypothesis(attemptsLog: string[]): string`
    - Per D-14: analyze pattern across attempts
    - If same test fails all cycles: "Same test failing across all cycles — likely a schema mismatch, missing migration, or incorrect storage function return type"
    - If different tests fail: "Different tests failing across cycles — likely a cascading issue from shared state or import error"
    - Never return empty string
