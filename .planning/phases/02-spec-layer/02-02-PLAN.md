export const PageSpecCore = z.object({ name: z.string().min(1), route: z.string().min(1), purpose: z.string().min(1), components: z.array(z.string()), authLevel: z.enum(["public", "authenticated", "admin"]), priority: z.number().int().min(1), dependsOn: z.array(z.string()).default([]), specVersion: z.number().int().default(1) });
export const PageSpecUI = z.object({ layoutHint: z.string().optional(), emptyState: z.string().optional(), loadingState: z.string().optional(), errorState: z.string().optional(), mobileConsiderations: z.string().optional() });
export const PageSpecData = z.object({ dataRequirements: z.array(z.object({ component: z.string(), fields: z.array(z.string()), source: z.string().optional() })), apiEndpoints: z.array(z.string()).default([]), validationRules: z.array(z.string()).default([]) });
export const PageSpecAnalytics = z.object({ events: z.array(z.object({ name: z.string(), trigger: z.string(), properties: z.array(z.string()).default([]) })).default([]), featureFlagCandidates: z.array(z.string()).default([]) });
export const PageSpecFull = PageSpecCore.merge(PageSpecUI).merge(PageSpecData).merge(PageSpecAnalytics);
export const SharedComponentSpec = z.object({ id: z.string().min(1), name: z.string().min(1), purpose: z.string(), usedByPages: z.array(z.string()), props: z.array(z.string()).default([]) });
export const BackendEndpointSpec = z.object({ method: z.enum(["GET", "POST", "PUT", "PATCH", "DELETE"]), path: z.string().min(1), description: z.string(), requestBody: z.array(z.string()).default([]), responseFields: z.array(z.string()).default([]), authRequired: z.boolean().default(true), uiPageRef: z.string().optional() });
export const BackendSpecSchema = z.object({ endpoints: z.array(BackendEndpointSpec), drizzleTableHints: z.array(z.string()).default([]), backgroundJobs: z.array(z.string()).default([]), mismatches: z.array(z.string()).default([]) });
export const SpecOutputSchema = z.object({ pages: z.array(PageSpecFull).min(1), sharedComponents: z.array(SharedComponentSpec).default([]), suggestedOrder: z.array(z.string()).default([]), backendSpec: BackendSpecSchema.optional() });
export type PageSpecFull = z.infer<typeof PageSpecFull>;
export type SpecOutput = z.infer<typeof SpecOutputSchema>;
export type BackendSpec = z.infer<typeof BackendSpecSchema>;
